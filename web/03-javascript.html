3JavaScript — логика и интерактив

JavaScript — язык программирования, который работает в браузере. Он делает страницу живой: реагирует на клики, меняет контент без перезагрузки, общается с сервером.

Зачем JS на сайте
  • Интерактив: кнопки, меню, слайдеры, табы, аккордеоны
  • Валидация форм: проверка email без отправки на сервер
  • Динамика: обновление данных без перезагрузки страницы
  • Хранение: сохранение настроек (тёмная тема, избранное)
  • Общение с API: загрузка курсов валют, погоды, новостей
  • Анимации: плавные переходы, эффекты появления

3.1. Подключение JS

<!-- 1. В конце body (правильный способ) -->
<script src="script.js"></script>

<!-- 2. Внутри HTML (для одной страницы) -->
<script>
  console.log('Привет, мир!');
</script>

<!-- 3. С атрибутом defer (загружается параллельно, выполняется после HTML) -->
<script src="script.js" defer></script>
Первая команда для проверки — откройте консоль браузера (F12) и увидите сообщение.

3.7. Методы массивов (самое важное)

const numbers = [1, 2, 3, 4, 5];

// push / pop — добавить / удалить в конец
numbers.push(6);       // [1,2,3,4,5,6]
numbers.pop();         // [1,2,3,4,5]

// shift / unshift — начало массива
numbers.unshift(0);    // [0,1,2,3,4,5]
numbers.shift();       // [1,2,3,4,5]

// map — преобразовать каждый элемент
const doubled = numbers.map(n => n * 2);
console.log(doubled);  // [2,4,6,8,10]

// filter — оставить только подходящие
const even = numbers.filter(n => n % 2 === 0);
console.log(even);     // [2,4]

// find — найти первый подходящий
const found = numbers.find(n => n > 3);
console.log(found);    // 4

// includes — есть ли элемент
console.log(numbers.includes(3));  // true

// reduce — свернуть в одно значение
const sum = numbers.reduce((acc, n) => acc + n, 0);
console.log(sum);      // 15

// sort — сортировка
const unsorted = [3, 1, 4, 1, 5];
unsorted.sort((a, b) => a - b);   // [1,1,3,4,5]
unsorted.sort((a, b) => b - a);   // [5,4,3,1,1]

// join — объединить в строку
console.log(numbers.join(', '));   // '1, 2, 3, 4, 5'
map, filter, reduce — три главных метода. Зная их, можно решить 90% задач с массивами.

3.11. Fetch — общение с сервером

// Простой GET-запрос
fetch('https://api.example.com/users')
  .then(response => response.json())
  .then(data => console.log(data))
  .catch(error => console.error('Ошибка:', error));

// Современный синтаксис async/await
async function loadUsers() {
  try {
    const response = await fetch('https://api.example.com/users');
    if (!response.ok) throw new Error('Ошибка запроса');
    const data = await response.json();
    console.log(data);
  } catch (error) {
    console.error('Ошибка:', error);
  }
}

// Реальный пример: курс валют ЦБ РФ
async function getCurrency() {
  const response = await fetch('https://www.cbr-xml-daily.ru/daily_json.js');
  const data = await response.json();
  const usd = data.Valute.USD.Value;
  console.log(`Курс доллара: ${usd} руб.`);
}
getCurrency();

// POST-запрос (отправка данных)
async function sendForm(formData) {
  const response = await fetch('/api/submit', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(formData)
  });
  return await response.json();
}

Пример 3. Список задач (TodoList)

// HTML:
// <input type="text" class="todo-input">
// <button class="todo-add">Добавить</button>
// <ul class="todo-list"></ul>

const input = document.querySelector('.todo-input');
const addBtn = document.querySelector('.todo-add');
const list = document.querySelector('.todo-list');

let todos = JSON.parse(localStorage.getItem('todos')) || [];

function render() {
  list.innerHTML = '';
  todos.forEach((todo, index) => {
    const li = document.createElement('li');
    li.textContent = todo;
    li.addEventListener('click', () => removeTodo(index));
    list.appendChild(li);
  });
}

function addTodo() {
  const text = input.value.trim();
  if (!text) return;
  todos.push(text);
  input.value = '';
  save();
  render();
}

function removeTodo(index) {
  todos.splice(index, 1);
  save();
  render();
}

function save() {
  localStorage.setItem('todos', JSON.stringify(todos));
}

addBtn.addEventListener('click', addTodo);
input.addEventListener('keydown', e => {
  if (e.key === 'Enter') addTodo();
});

render();

Пример 4. Валидация формы

const form = document.querySelector('form');
const emailInput = document.querySelector('#email');
const errorMsg = document.querySelector('.error');

form.addEventListener('submit', (e) => {
  e.preventDefault();
  const email = emailInput.value.trim();

  // Проверка на пустоту
  if (!email) {
    errorMsg.textContent = 'Введите email';
    return;
  }

  // Проверка через regex
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  if (!emailRegex.test(email)) {
    errorMsg.textContent = 'Неверный формат email';
    return;
  }

  // Всё ок
  errorMsg.textContent = '';
  alert('Форма отправлена!');
  form.reset();
});

Вопросы для самопроверки

Вопрос 1. Чем let отличается от const?
let — переменная, значение можно переприсвоить: 
let x = 5;
x = 10;  // OK
const — константа, значение менять нельзя: 
const y = 5;
y = 10;  // Ошибка!
Но: если const — это объект или массив, можно менять его свойства: 
const user = { name: 'Иван' };
user.name = 'Пётр';  // OK — меняем свойство, а не саму константу
Рекомендация: используйте const по умолчанию. let — только если значение точно будет меняться.
Вопрос 2. Что такое DOM простыми словами?
DOM — это представление HTML-страницы в виде дерева объектов, которыми можно управлять из JavaScript.

Когда браузер загружает HTML, он превращает его в DOM-дерево. Каждый тег — узел дерева. Например: 
<body>
  <h1>Заголовок</h1>
  <p>Текст</p>
</body>
Превращается в: 
body
├── h1
│   └── "Заголовок"
└── p
    └── "Текст"
JS может найти любой узел (querySelector), изменить его (textContent), добавить новый (createElement) или удалить (remove). Это и есть «управление DOM».
Вопрос 3. Почему === лучше ==?
== — нестрогое сравнение, автоматически приводит типы: 
5 == '5'    // true (строка превращается в число)
0 == false  // true
null == undefined // true
'' == 0     // true
=== — строгое сравнение, проверяет и значение, и тип: 
5 === '5'    // false (разные типы)
0 === false  // false
null === undefined // false
'' === 0     // false
Правило: всегда используйте === и !==. Автоприведение типов даёт неожиданные баги.
Вопрос 4. Что такое async/await и зачем он нужен?
Проблема: некоторые операции занимают время — запрос к серверу, загрузка файла. Пока они выполняются, JS не должен «зависать».

Решение: асинхронные функции. Ключевые слова:
  • async — помечает функцию как асинхронную. Она всегда возвращает промис.
  • await — «подожди, пока операция завершится». Можно использовать только внутри async-функции.
Пример: 
async function getData() {
  const response = await fetch('https://api.example.com/data');
  const data = await response.json();
  console.log(data);
}
Без await код пошёл бы дальше, не дождавшись ответа. С await — «остановись здесь, дождись, потом продолжай».

Аналогия: обычный код — «заказал пиццу и ждёшь у двери». Async — «заказал пиццу и пошёл заниматься своими делами, вернулся, когда её привезли».
