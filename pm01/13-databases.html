12Теория баз данных и программирование MS SQL Server

База данных (БД) — организованное хранилище структурированных данных. СУБД (система управления базами данных) — программа, которая этими данными управляет. SQL — язык запросов, стандарт для работы с реляционными БД.

В этом курсе — классическая реляционная модель на примере MS SQL Server.

12.2. Реляционная модель

Таблица — основной объект. Состоит из строк (записей) и столбцов (полей).

Связь «один ко многим»: у одного пользователя много заказов (user_id ссылается на Users.id).

Ключевые понятия

ПонятиеЧто это
Первичный ключ (PK)Уникальный идентификатор строки. Не может быть NULL, не повторяется.
Внешний ключ (FK)Ссылка на PK другой таблицы. Обеспечивает ссылочную целостность.
ИндексСтруктура для ускорения поиска (как оглавление в книге).
НормализацияУстранение дублирования данных через разделение на таблицы.
ТранзакцияНабор операций, которые выполняются «всё или ничего» (ACID).

12.4. Базовые SQL-запросы

SELECT — выборка

-- Все столбцы, все строки
SELECT * FROM Users;

-- Конкретные столбцы
SELECT id, name FROM Users;

-- С условием
SELECT * FROM Users WHERE age >= 18;

-- С сортировкой (ASC — по возрастанию, DESC — по убыванию)
SELECT * FROM Users ORDER BY name ASC;

-- С ограничением количества
SELECT TOP 10 * FROM Users ORDER BY registered_at DESC;

-- С уникальными значениями
SELECT DISTINCT city FROM Users;

-- С псевдонимами
SELECT name AS user_name, age AS user_age FROM Users;

-- Комбинированное условие
SELECT * FROM Users
WHERE age >= 18 AND city = 'Москва'
ORDER BY name;

INSERT, UPDATE, DELETE

-- Вставка одной строки
INSERT INTO Users (name, age, email)
VALUES ('Alice', 25, 'alice@example.com');

-- Вставка нескольких строк
INSERT INTO Users (name, age, email) VALUES
    ('Bob', 30, 'bob@example.com'),
    ('Carol', 22, 'carol@example.com');

-- Обновление
UPDATE Users
SET age = 26, city = 'СПб'
WHERE name = 'Alice';

-- Удаление
DELETE FROM Users WHERE age < 18;

-- Осторожно! Без WHERE удалит всё
-- DELETE FROM Users;

-- Безопаснее — сначала проверить через SELECT
SELECT * FROM Users WHERE age < 18;
⚠ Всегда с WHERE
UPDATE и DELETE без WHERE затронут все строки таблицы. Всегда пишите WHERE. Привычка: сначала SELECT с тем же WHERE, посмотреть, что попадает под условие, потом заменить на UPDATE/DELETE.

12.5. JOIN — соединение таблиц

Главная сила SQL — соединение данных из разных таблиц по ключам.

INNER JOIN возвращает только совпадающие строки из обеих таблиц.
-- INNER JOIN: только совпадения
SELECT u.name, o.sum
FROM Users u
INNER JOIN Orders o ON o.user_id = u.id;

-- LEFT JOIN: все из левой + совпадения из правой
SELECT u.name, o.sum
FROM Users u
LEFT JOIN Orders o ON o.user_id = u.id;
-- Пользователи без заказов тоже попадут (с NULL в sum)

-- RIGHT JOIN: все из правой
SELECT u.name, o.sum
FROM Users u
RIGHT JOIN Orders o ON o.user_id = u.id;

-- FULL OUTER JOIN: все из обеих
SELECT u.name, o.sum
FROM Users u
FULL OUTER JOIN Orders o ON o.user_id = u.id;

-- JOIN трёх таблиц
SELECT u.name, o.id AS order_id, p.title
FROM Users u
JOIN Orders o ON o.user_id = u.id
JOIN Products p ON p.id = o.product_id;

12.6. Агрегатные функции и GROUP BY

-- Агрегаты
SELECT COUNT(*) FROM Users;              -- количество строк
SELECT AVG(age) FROM Users;              -- среднее
SELECT MIN(age), MAX(age) FROM Users;    -- мин и макс
SELECT SUM(sum) FROM Orders;             -- сумма

-- GROUP BY: агрегация по группам
SELECT city, COUNT(*) AS user_count
FROM Users
GROUP BY city;

SELECT user_id, SUM(sum) AS total_spent
FROM Orders
GROUP BY user_id
ORDER BY total_spent DESC;

-- HAVING: фильтр ПОСЛЕ группировки
SELECT city, COUNT(*) AS user_count
FROM Users
GROUP BY city
HAVING COUNT(*) > 10;   -- только города с >10 юзерами

-- Разница: WHERE фильтрует ДО группировки, HAVING — после
SELECT city, COUNT(*) AS cnt
FROM Users
WHERE age >= 18
GROUP BY city
HAVING COUNT(*) > 5;

12.7. Создание таблиц (DDL)

-- Создание таблицы
CREATE TABLE Users (
    id INT IDENTITY(1,1) PRIMARY KEY,
    name NVARCHAR(100) NOT NULL,
    email NVARCHAR(200) UNIQUE NOT NULL,
    age INT CHECK (age >= 0 AND age <= 150),
    city NVARCHAR(100),
    registered_at DATETIME2 DEFAULT GETDATE()
);

CREATE TABLE Orders (
    id INT IDENTITY(1,1) PRIMARY KEY,
    user_id INT NOT NULL,
    sum DECIMAL(10, 2) NOT NULL,
    created_at DATETIME2 DEFAULT GETDATE(),
    CONSTRAINT FK_Orders_Users
        FOREIGN KEY (user_id)
        REFERENCES Users(id)
        ON DELETE CASCADE
);

-- Индексы для ускорения поиска
CREATE INDEX IX_Users_Email ON Users(email);
CREATE INDEX IX_Orders_UserId ON Orders(user_id);

-- Изменение структуры
ALTER TABLE Users ADD phone NVARCHAR(20);
ALTER TABLE Users DROP COLUMN city;

-- Удаление
DROP TABLE Orders;
DROP TABLE Users;   -- сначала зависимые таблицы!

12.8. Хранимые процедуры и функции

-- Хранимая процедура: добавить пользователя
CREATE PROCEDURE AddUser
    @name NVARCHAR(100),
    @email NVARCHAR(200),
    @age INT
AS
BEGIN
    IF EXISTS (SELECT 1 FROM Users WHERE email = @email)
    BEGIN
        THROW 50000, 'Email уже существует', 1;
    END

    INSERT INTO Users (name, email, age)
    VALUES (@name, @email, @age);

    SELECT SCOPE_IDENTITY() AS new_id;
END;

-- Вызов
EXEC AddUser @name = 'Alice', @email = 'a@b.com', @age = 25;

-- Функция: подсчёт заказов пользователя
CREATE FUNCTION GetUserOrderCount(@userId INT)
RETURNS INT
AS
BEGIN
    DECLARE @count INT;
    SELECT @count = COUNT(*) FROM Orders WHERE user_id = @userId;
    RETURN @count;
END;

-- Использование
SELECT dbo.GetUserOrderCount(1);

12.9. Триггеры

Триггер — процедура, которая автоматически выполняется при INSERT/UPDATE/DELETE.

-- Триггер: логировать изменение цены
CREATE TABLE PriceLog (
    id INT IDENTITY(1,1) PRIMARY KEY,
    product_id INT,
    old_price DECIMAL(10,2),
    new_price DECIMAL(10,2),
    changed_at DATETIME2 DEFAULT GETDATE()
);

CREATE TRIGGER trg_Products_PriceChange
ON Products
AFTER UPDATE
AS
BEGIN
    INSERT INTO PriceLog (product_id, old_price, new_price)
    SELECT d.id, d.price, i.price
    FROM deleted d
    JOIN inserted i ON d.id = i.id
    WHERE d.price <> i.price;
END;

12.10. Транзакции

-- Транзакция: перевод денег между счетами
BEGIN TRANSACTION;

BEGIN TRY
    -- Списать с одного
    UPDATE Accounts SET balance = balance - 1000 WHERE id = 1;

    -- Проверка: не ушли ли в минус
    IF (SELECT balance FROM Accounts WHERE id = 1) < 0
        THROW 50000, 'Недостаточно средств', 1;

    -- Зачислить другому
    UPDATE Accounts SET balance = balance + 1000 WHERE id = 2;

    COMMIT TRANSACTION;
END TRY
BEGIN CATCH
    ROLLBACK TRANSACTION;
    THROW;
END CATCH;

-- Уровни изоляции (важно для многопользовательских систем)
SET TRANSACTION ISOLATION LEVEL READ COMMITTED;  -- по умолчанию
-- READ UNCOMMITTED — читает «грязные» данные
-- REPEATABLE READ — блокирует строки до конца транзакции
-- SERIALIZABLE — полная изоляция, медленно

12.11. Нормализация

Нормализация — процесс устранения избыточности и аномалий в схеме БД. Три основные нормальные формы:

ФормаТребование
1NFВсе значения атомарны (не списки, не множества)
2NF1NF + нет частичных зависимостей от составного ключа
3NF2NF + нет транзитивных зависимостей (не-ключ → не-ключ)
Пример нормализации
Плохо (1 таблица): 
Orders: id | user_name | user_email | product_title | product_price | quantity
Проблемы: дублирование данных о пользователе и товаре; при изменении email — обновлять во всех заказах.

Хорошо (3 таблицы, 3NF): 
Users:    id | name | email
Products: id | title | price
Orders:   id | user_id | product_id | quantity | created_at
Нет дублирования, есть ссылочная целостность.

12.12. Программирование: работа с БД из C#

ADO.NET — низкоуровневый доступ

using System.Data.SqlClient;

const string connectionString = 
    "Server=localhost;Database=ShopDb;Trusted_Connection=True;";

// SELECT
using (var connection = new SqlConnection(connectionString))
{
    connection.Open();
    var cmd = new SqlCommand("SELECT id, name FROM Users WHERE age > @age", connection);
    cmd.Parameters.AddWithValue("@age", 18);

    using (var reader = cmd.ExecuteReader())
    {
        while (reader.Read())
        {
            int id = reader.GetInt32(0);
            string name = reader.GetString(1);
            Console.WriteLine($"{id}: {name}");
        }
    }
}

// INSERT с параметрами (защита от SQL-инъекций!)
using (var connection = new SqlConnection(connectionString))
{
    connection.Open();
    var cmd = new SqlCommand(
        "INSERT INTO Users (name, email) VALUES (@name, @email)", connection);
    cmd.Parameters.AddWithValue("@name", userInput.Name);
    cmd.Parameters.AddWithValue("@email", userInput.Email);
    int rows = cmd.ExecuteNonQuery();
}
⚠ SQL-инъекция
Никогда не склеивайте SQL-строку из пользовательского ввода: 
// ПЛОХО — дыра в безопасности!
var sql = "SELECT * FROM Users WHERE name = '" + userInput + "'";
// Если userInput = "'; DROP TABLE Users; --" — прощай, таблица
Всегда используйте параметры: 
cmd.Parameters.AddWithValue("@name", userInput);

Entity Framework — ORM

ORM (Object-Relational Mapping) — библиотека, которая превращает таблицы БД в C#-классы.

// Модели
public class User {
    public int Id { get; set; }
    public string Name { get; set; }
    public string Email { get; set; }
    public List<Order> Orders { get; set; }
}

public class Order {
    public int Id { get; set; }
    public int UserId { get; set; }
    public User User { get; set; }
    public decimal Sum { get; set; }
}

// DbContext
public class ShopContext : DbContext {
    public DbSet<User> Users { get; set; }
    public DbSet<Order> Orders { get; set; }

    protected override void OnConfiguring(DbContextOptionsBuilder options) {
        options.UseSqlServer("Server=localhost;Database=ShopDb;Trusted_Connection=True;");
    }
}

// Использование
using var db = new ShopContext();

// SELECT с LINQ
var adults = db.Users.Where(u => u.Age >= 18).ToList();
var bigOrders = db.Orders.Where(o => o.Sum > 1000).OrderByDescending(o => o.Sum).ToList();

// JOIN через навигационные свойства
var usersWithOrders = db.Users
    .Include(u => u.Orders)
    .Where(u => u.Orders.Any())
    .ToList();

// INSERT
db.Users.Add(new User { Name = "Alice", Email = "alice@example.com" });
db.SaveChanges();

// UPDATE
var user = db.Users.Find(1);
user.Name = "New Name";
db.SaveChanges();

// DELETE
db.Users.Remove(user);
db.SaveChanges();

Вопросы для самопроверки

Вопрос 1. Что такое SQL-инъекция и как от неё защититься?
SQL-инъекция — атака, когда злоумышленник вставляет SQL-код через пользовательский ввод.

Пример атаки: 
// Код приложения:
var sql = "SELECT * FROM Users WHERE name = '" + input + "'";

// Пользователь вводит:
// ' OR '1'='1
// Итоговый запрос:
// SELECT * FROM Users WHERE name = '' OR '1'='1'
// Возвращает ВСЕ строки!
Ещё опаснее: 
// Пользователь вводит: '; DROP TABLE Users; --
// Удаляется вся таблица
Защита: параметризованные запросы: 
cmd.CommandText = "SELECT * FROM Users WHERE name = @name";
cmd.Parameters.AddWithValue("@name", userInput);
Параметры передаются отдельно от SQL-кода, поэтому никогда не интерпретируются как SQL.
Вопрос 2. Чем WHERE отличается от HAVING?
WHERE — фильтрует строки до группировки.
HAVING — фильтрует группы после группировки.

Порядок выполнения запроса:
  1. FROM — берём таблицу
  2. WHERE — фильтруем строки
  3. GROUP BY — группируем
  4. HAVING — фильтруем группы
  5. SELECT — выбираем столбцы
  6. ORDER BY — сортируем
Пример: 
SELECT city, COUNT(*) AS cnt
FROM Users
WHERE age >= 18          -- фильтр строк (до группировки)
GROUP BY city
HAVING COUNT(*) > 5;     -- фильтр групп (после)
Вопрос 3. Зачем нужны транзакции?
Транзакция гарантирует атомарность: либо все операции выполняются, либо ни одна.

Классический пример — перевод денег:
  1. Списать 1000 руб. со счёта A
  2. Зачислить 1000 руб. на счёт B
Если между шагами 1 и 2 упадёт сервер — деньги исчезнут. Транзакция гарантирует: если что-то упало, откат к исходному состоянию.

Свойства ACID:
  • Atomicity — всё или ничего
  • Consistency — данные не нарушают правила (FK, CHECK)
  • Isolation — параллельные транзакции не мешают друг другу
  • Durability — после COMMIT данные сохранены, даже если отключится питание
Вопрос 4. Что выбрать: EF, Dapper или ADO.NET?
ADO.NET — когда:
  • Нужен максимальный контроль над SQL
  • Критична производительность (миллионы запросов в секунду)
  • Простые сценарии, не хочется тащить ORM
Dapper — когда:
  • Есть готовые SQL-запросы, нужен маппинг в объекты
  • Read-heavy проекты (аналитика, отчёты)
  • Хотите скорость ADO.NET + удобство
Entity Framework — когда:
  • Сложная доменная модель (много сущностей, связей)
  • Нужны миграции схемы
  • Команда знает LINQ
  • Хочется избежать ручного SQL
В реальных проектах часто комбинируют: EF для CRUD, Dapper для отчётов.
